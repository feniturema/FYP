# Reproduction of the review finding: a SKIP LOCKED publisher batch gap-locks hot-path outbox INSERTs
# under REPEATABLE READ (MySQL default) but not under READ COMMITTED. mysql:8.0.46, schema = V2 order_outbox,
# rows: 2 SENT + 1 NEW. Session A holds the batch for 6 s; session B inserts a NEW row with lock wait 3 s.

publisher isolation=REPEATABLE READ -> hot-path INSERT: ERROR 1205 (HY000) at line 1: Lock wait timeout exceeded; try restarting transaction (waited 3s)
publisher isolation=READ COMMITTED -> hot-path INSERT: insert ok (waited 0s)

# script:
    set -u
    docker run -d --name ftsm-p2-gaplock --label ftsm.phase=p2 -e MYSQL_ROOT_PASSWORD=root -e MYSQL_DATABASE=t mysql:8.0.46 >/dev/null
    for i in $(seq 1 60); do docker exec ftsm-p2-gaplock mysqladmin -uroot -proot ping >/dev/null 2>&1 && docker exec ftsm-p2-gaplock mysql -uroot -proot -e 'select 1' t >/dev/null 2>&1 && break; sleep 2; done
    M() { docker exec -i ftsm-p2-gaplock mysql -uroot -proot -N t 2>/dev/null; }
    M <<'SQL'
    CREATE TABLE order_outbox (id BIGINT AUTO_INCREMENT PRIMARY KEY, order_id CHAR(36) NOT NULL, user_id BIGINT NOT NULL,
     event_id BIGINT NOT NULL, payload JSON NOT NULL, status TINYINT NOT NULL DEFAULT 0, created_at DATETIME(3) NOT NULL DEFAULT CURRENT_TIMESTAMP(3),
     sent_at DATETIME(3) NULL, UNIQUE KEY uk_outbox_order_id (order_id), KEY idx_outbox_status_id (status, id), KEY idx_outbox_event_status (event_id, status));
    INSERT INTO order_outbox(order_id,user_id,event_id,payload,status) VALUES ('sent-1',1,1,'{}',1),('sent-2',2,1,'{}',1),('new-1',3,1,'{}',0);
    SQL
    for iso in 'REPEATABLE READ' 'READ COMMITTED'; do
      # session A: publisher batch holding its locks for 6 s
      ( echo "SET SESSION TRANSACTION ISOLATION LEVEL $iso; BEGIN; SELECT id FROM order_outbox WHERE status=0 ORDER BY id LIMIT 500 FOR UPDATE SKIP LOCKED; DO SLEEP(6); ROLLBACK;" | M >/dev/null ) &
      sleep 1.5
      # session B: hot-path autocommit INSERT of a new NEW row (default isolation, lock wait 3 s)
      t0=$(date +%s)
      out=$(echo "SET SESSION innodb_lock_wait_timeout=3; INSERT INTO order_outbox(order_id,user_id,event_id,payload) VALUES (UUID(),9,1,'{}'); SELECT 'insert ok';" | docker exec -i ftsm-p2-gaplock mysql -uroot -proot -N t 2>&1 | grep -v Warning)
      echo "publisher isolation=$iso -> hot-path INSERT: ${out:-?} (waited $(( $(date +%s) - t0 ))s)"
      wait
    done
    docker rm -f ftsm-p2-gaplock >/dev/null
